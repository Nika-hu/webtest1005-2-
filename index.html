const ADMIN_PASSWORD = "Hess_PBDadmin"; function doGet(e) { try { const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0]; const data = sheet.getDataRange().getValues(); if (data.length <= 1) { return responseJSON({ status: 'success', data: [] }); } const headers = data[0]; const rows = data.slice(1); const result = rows.map(row => { let obj = {}; headers.forEach((h, i) => { let val = row[i]; if (h === 'tags' || h === 'skills') { try { val = JSON.parse(val); } catch(e) { val = String(val).split(',').map(s=>s.trim()).filter(Boolean); } } if (h === 'isExpired') { val = (String(val).toLowerCase() === 'true' || val === true); } obj[h] = val; }); return obj; }); return responseJSON({ status: 'success', data: result }); } catch (err) { return responseJSON({ status: 'error', message: err.toString() }); } } function doPost(e) { try { const contents = JSON.parse(e.postData.contents); const action = contents.action; const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheets()[0]; // 驗證後台密碼 if (action === 'verifyPassword') { if (contents.password === ADMIN_PASSWORD) { return responseJSON({ status: 'success' }); } else { return responseJSON({ status: 'error', message: '密碼不正確' }); } } // 新增研習場次 if (action === 'create') { const item = contents.data; const newRow = [ item.id, item.title, item.mode, item.date, item.speaker, item.deadline, JSON.stringify(item.tags || []), item.location, item.googleFormUrl, JSON.stringify(item.skills || []), item.image, item.isExpired ]; sheet.appendRow(newRow); return responseJSON({ status: 'success' }); } // 修改或刪除研習場次 if (action === 'update' || action === 'delete') { const data = sheet.getDataRange().getValues(); const idToMatch = Number(action === 'update' ? contents.data.id : contents.id); let targetRowIndex = -1; for (let i = 1; i < data.length; i++) { if (Number(data[i][0]) === idToMatch) { targetRowIndex = i + 1; break; } } if (targetRowIndex === -1) { return responseJSON({ status: 'error', message: '在試算表中找不到對應的 ID' }); } if (action === 'delete') { sheet.deleteRow(targetRowIndex); return responseJSON({ status: 'success' }); } if (action === 'update') { const item = contents.data; const updatedRow = [ item.id, item.title, item.mode, item.date, item.speaker, item.deadline, JSON.stringify(item.tags || []), item.location, item.googleFormUrl, JSON.stringify(item.skills || []), item.image, item.isExpired ]; sheet.getRange(targetRowIndex, 1, 1, updatedRow.length).setValues([updatedRow]); return responseJSON({ status: 'success' }); } } return responseJSON({ status: 'error', message: '未知的指令' }); } catch (err) { return responseJSON({ status: 'error', message: err.toString() }); } } function responseJSON(data) { return ContentService.createTextOutput(JSON.stringify(data)) .setMimeType(ContentService.MimeType.JSON); }
